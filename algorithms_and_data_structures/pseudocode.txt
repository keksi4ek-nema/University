def поиск_рек(A, ключ, i=0):
    # Базовый случай: дошли до конца списка — не найдено
    if i == len(A):
        return -1
    # Базовый случай: нашли элемент
    if A[i] == ключ:
        return i
    # Рекурсивный вызов для следующего индекса
    return поиск_рек(A, ключ, i + 1)




def поиск(состояние):
    # Базовый случай: если текущее состояние — решение
    if является_решением(состояние):
        return True

    # Перебираем все допустимые ходы из текущего состояния
    for ход in допустимые_ходы(состояние):
        # Применяем ход
        применить_ход(состояние, ход)

        # Рекурсивно проверяем новое состояние
        if поиск(состояние):
            return True

        # Отменяем ход (возврат к предыдущему состоянию — backtrack)
        отменить_ход(состояние, ход)

    # Если ни один из ходов не привёл к решению
    return False

функция Fib(n):
    если n <= 1:
        вернуть n
    вернуть Fib(n - 1) + Fib(n - 2)


функция FibМемо(n, memo):
    если n <= 1:
        вернуть n

    если memo[n] != -1:
        вернуть memo[n]

    memo[n] = FibМемо(n - 1, memo) + FibМемо(n - 2, memo)
    вернуть memo[n]
    


функция FibТабл(n):
    если n <= 1:
        вернуть n

    создать массив dp размера n + 1
    dp[0] = 0
    dp[1] = 1

    для i от 2 до n:
        dp[i] = dp[i - 1] + dp[i - 2]

    вернуть dp[n]

функция Обход(x, y, шаг):
    если шаг == N * N:
        вернуть True

    кандидаты = пустой список
    для каждого направления k из 8 возможных:
        nx = x + dx[k]
        ny = y + dy[k]
        если nx и ny внутри доски И доска[nx][ny] == -1:
            добавить (nx, ny) в кандидаты

    # Эвристика Варнсдорфа: сортируем кандидатов по возрастанию числа их свободных ходов
    для каждой клетки (nx, ny) из кандидатов:
        free = число непосещённых клеток, достижимых из (nx, ny)
        сохранить free вместе с (nx, ny)
    отсортировать кандидаты по значению free (по возрастанию)

    для каждой клетки (nx, ny) из кандидатов:
        доска[nx][ny] = шаг          # сделать ход
        если Обход(nx, ny, шаг + 1):
            вернуть True
        доска[nx][ny] = -1           # откат (backtrack)

    вернуть False

функция ФакторРек(n, d):
    если n < 2:
        вернуть пустой список []

    если d * d > n:
        // Дальше искать делители нет смысла: оставшийся n — простое число
        вернуть список [n]

    если n % d == 0:
        // d — делитель: добавляем его и продолжаем делить оставшуюся часть
        вернуть [d] + ФакторРек(n // d, d)

    // d не делит n: пробуем следующий кандидат
    вернуть ФакторРек(n, d + 1)



функция ФакторИтер(n):
    результат = []
    d = 2

    пока d * d <= n:
        пока n % d == 0:
            добавить d в результат
            n = n // d          // целочисленное деление
        d = d + 1

    если n > 1:
        добавить n в результат

    вернуть результат



функция Разложить(n, spf):
    результат = []

    пока n > 1:
        p = spf[n]                  // берём наименьший простой делитель из таблицы
        добавить p в результат
        n = n // p                  // делим n на этот делитель (целочисленное деление)

    вернуть результат
